// From https://www.youtube.com/watch?v=9h_61hXCDuI)

def print_obj(obj, tag=None):
    """
    Print an object in terms of type and then display is.
    """
    if tag:
        print(tag)
    print("type=", type(obj))
    display(obj)


def print_dask(obj, visualize_graph=True, compute=True):
    """
    Print information about a dask task graph.
    """
    print("type=", type(obj))
    print("obj=", obj)
    #
    print("# display")
    display(obj)
    #
    print("# dask")
    display(obj.dask)
    #
    if visualize_graph:
        print("# visualize")
        display(obj.visualize())
    #
    if compute:
        print("# compute")
        res = obj.compute()
        print(type(res))
        print(res)

import numpy as np


def load(x):
    """
    Get a sizable array.
    """
    time.sleep(0.2)
    return np.arange(1_000_000) + x


def process(x):
    time.sleep(0.1)
    return x + 1


def save(x):
    time.sleep(0.4)
%%time

inputs = range(50)

for i in inputs:
    x = load(i)
    y = process(x)
    save(y)

# It takes (0.2 + 0.1 + 0.4) * 50 = 0.7 * 50 = 35s
