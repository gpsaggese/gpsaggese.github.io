  \end{itemize}

  \textbf{\textcolor{red}{< 2000s}}

  \textbf{\textcolor{orange}{Pros}}

  \item

  \end{itemize}

  \textbf{\textcolor{darkgreen}{Cons}}

  \begin{itemize}

      networks

  \item

  \item

  \begin{itemize}

  \item
    Stacks layers as a unified filesystem
  \item

  \item
    Files from top layers can obscure files from bottom layers
  \end{itemize}

  \textbf{\textcolor{blue}{Layer hash}}

  \begin{itemize}
  \tightlist
  \item
    Each layer has a hash based on content
  \item
    Layers are pulled and pushed in a compressed format
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Image hash}}

  \begin{itemize}
  \tightlist
  \item
    Each image has a hash
  \item
    Hash is a function of the config file and layers
  \item
    Image changes generate a new hash
  \end{itemize}
\end{itemize}

\begin{frame}{Docker: Container Data}
\protect\phantomsection\label{docker-container-data}
\begin{itemize}
\item
  A container has access to different data
\item
  \textbf{\textcolor{red}{Container storage}}

  \begin{itemize}
  \tightlist
  \item
    Copy-on-write layer in the image
  \item
    Ephemeral (temporary data)
  \item
    Data persists until the container is removed

    \begin{itemize}
    \tightlist
    \item
      Stopping or pausing a container doesn't lose data
    \end{itemize}
  \item
    Containers are immutable

    \begin{itemize}
    \tightlist
    \item
      Avoid writing persistent data into containers
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Bind-mount a local dir}}

  \begin{itemize}
  \tightlist
  \item
    Mount a local directory to a directory inside a container
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Docker volumes}}

  \begin{itemize}
  \tightlist
  \item
    Volumes exist separately from the container

    \begin{itemize}
    \tightlist
    \item
      E.g., store PostgreSQL DB content
    \end{itemize}
  \item
    State is permanent across container invocations
  \item
    Shareable across containers
  \end{itemize}
\end{itemize}
\end{frame}
