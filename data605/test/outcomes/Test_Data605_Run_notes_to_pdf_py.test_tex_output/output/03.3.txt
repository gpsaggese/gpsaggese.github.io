\begin{frame}{Docker Compose}
\protect\phantomsection\label{docker-compose}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Manage multi-container apps running on a single node}}

  \begin{itemize}
  \tightlist
  \item
    Describe app in a single \emph{declarative} YAML file

    \begin{itemize}
    \tightlist
    \item
      Avoid long Docker command scripts
    \end{itemize}
  \item
    Compose interacts with Docker API to achieve the requested state
  \item
    Examples:

    \begin{itemize}
    \tightlist
    \item
      Client app and Postgres DB
    \item
      Microservices: Web front-end, Ordering, Back-end DB
    \end{itemize}
  \item
    In 2020, Docker Compose became an open standard for
    ``code-to-cloud'' process
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Manage multi-container apps running on multiple hosts}}

  \begin{itemize}
  \tightlist
  \item
    Docker Stacks / Swarm
  \item
    Kubernetes
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Docker Compose: Tutorial Example}
\protect\phantomsection\label{docker-compose-tutorial-example}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  Default name for a Compose file is
  \textcolor{blue}{\texttt{docker-compose.yml}}

  \begin{itemize}
  \tightlist
  \item
    Specify \textcolor{blue}{\texttt{-f}} for custom filenames
  \end{itemize}
\item
  \textbf{\textcolor{red}{Top-level keys}}:

  \begin{itemize}
  \tightlist
  \item
    \textcolor{blue}{\texttt{version}}:

    \begin{itemize}
    \tightlist
    \item
      Mandatory first line for API version
    \item
      Use latest version, typically 3 or higher
    \end{itemize}
  \item
    \textcolor{blue}{\texttt{services}}:

    \begin{itemize}
    \tightlist
    \item
      Define microservices
    \end{itemize}
  \item

    \begin{itemize}

\end{verbatim}

  \endgroup

  Show the status of the service \begingroup \footnotesize \color{blue}

\begin{verbatim}
> docker compose ls
\end{verbatim}

  \endgroup

\begin{verbatim}
> docker compose up
\end{verbatim}

  \endgroup

  \begingroup \footnotesize \color{blue}

  \endgroup
