\begin{frame}{Git Branching}
\protect\phantomsection\label{git-branching}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Branching}}

  \begin{itemize}
  \tightlist
  \item
    Diverge from the main development line
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Why branch?}}

  \begin{itemize}
  \tightlist
  \item
    Work without affecting the main code
  \item
    Avoid changes in the main branch
  \item
    Merge code downstream for updates
  \item
    Merge code upstream after completion
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.4\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_2/lec_2_slide_15_image_1.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Git branching is lightweight}}

  \begin{itemize}
  \tightlist
  \item
    Instantaneous
  \item
    A branch is a pointer to a commit
  \item
    Git stores data as snapshots, not file differences
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Git workflows branch and merge often}}

  \begin{itemize}
  \tightlist
  \item
    Multiple times a day
  \item
    Surprising for users of centralized VCS

    \begin{itemize}
    \tightlist
    \item
      E.g., branch before lunch
    \end{itemize}
  \item
    Branches are cheap

    \begin{itemize}
    \tightlist
    \item
      Use them to isolate and organize work
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Git Branching}
\protect\phantomsection\label{git-branching-1}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  \textcolor{blue}{\texttt{master}} (or \textcolor{blue}{\texttt{main}})
  is a normal branch

  \begin{itemize}
  \tightlist
  \item
    Pointer to the last commit
  \item
    Moves forward with each commit
  \end{itemize}
\item
  \textcolor{blue}{\texttt{HEAD}}

  \begin{itemize}
  \tightlist
  \item
    Pointer to the current branch
  \item
    E.g., \textcolor{blue}{\texttt{master}},
    \textcolor{blue}{\texttt{testing}}
  \item
    \textcolor{blue}{\texttt{git checkout <BRANCH>}} moves across
    branches
  \end{itemize}
\item
  \textcolor{blue}{\texttt{git branch testing}}

  \begin{itemize}
  \tightlist
  \item
    Creates a new pointer \textcolor{blue}{\texttt{testing}}
  \item
    Points to the current commit
  \item
    Pointer is movable
  \end{itemize}
\item
  Divergent history

  \begin{itemize}
  \tightlist
  \item
    Work progresses in two ``split'' branches
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Git Checkout}
\protect\phantomsection\label{git-checkout}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  \textcolor{blue}{\texttt{git checkout}} switches branches

  \begin{itemize}
  \tightlist
  \item
    Moves \textcolor{blue}{\texttt{HEAD}} pointer to the new branch
  \item
    Changes files in the working directory to match the branch pointer
  \end{itemize}
\item
  E.g., two branches, \textcolor{blue}{\texttt{master}} and
  \textcolor{blue}{\texttt{testing}}

  \begin{itemize}
  \tightlist
  \item
    You are on \textcolor{blue}{\texttt{master}}
  \item
    \textcolor{blue}{\texttt{git checkout testing}}
  \item
    Pointer moves, working directory changes
  \item
    Keep working and commit on \textcolor{blue}{\texttt{testing}}
  \item
    Pointer to \textcolor{blue}{\texttt{testing}} moves forward
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}[fragile]{Git Rebasing}
\protect\phantomsection\label{git-rebasing}
\begin{itemize}
\tightlist
\item
  In Git, there are
  \textbf{\textcolor{red}{two ways of merging divergent history}}
\item
  E.g., consider \textcolor{blue}{\texttt{master}} and
  \textcolor{blue}{\texttt{experiment}} have a common ancestor
  \textcolor{blue}{\texttt{C2}}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Merge}}

  \begin{itemize}
  \item
    Go to the target branch \begingroup \small \color{blue}

\begin{verbatim}
> git checkout master
> git merge experiment
\end{verbatim}

    \endgroup
  \item
    Create a new snapshot \textcolor{blue}{\texttt{C5}} and commit
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Rebase}}

  \begin{itemize}
  \item
    Go to the branch to rebase \begingroup \small \color{blue}

\begin{verbatim}
> git checkout experiment
> git rebase master
\end{verbatim}

    \endgroup
  \item
    Rebase algorithm:

    \begin{itemize}
    \tightlist
    \item
      Get all the changes committed in the branch
      (\textcolor{blue}{\texttt{C4}}) where we are on
      (\textcolor{blue}{\texttt{experiment}}) since the common ancestor
      (\textcolor{blue}{\texttt{C2}})
    \item
      Sync to the branch that we are rebasing onto
      (\textcolor{blue}{\texttt{master}} at
      \textcolor{blue}{\texttt{C3}})
    \item
      Apply the changes \textcolor{blue}{\texttt{C4}}
    \item
      Only the current branch is affected
    \item
      Finally, fast forward \textcolor{blue}{\texttt{experiment}}
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Uses of Rebase}
\protect\phantomsection\label{uses-of-rebase}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Rebasing makes for a cleaner history}}

  \begin{itemize}
  \tightlist
  \item
    The history looks like all the work happened in series
  \item
    Although in reality, it happened in parallel to the development in
    the master branch
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Rebasing to contribute to a project}}

  \begin{itemize}
  \tightlist
  \item
    Developer

    \begin{itemize}
    \tightlist
    \item
      You are contributing to a project that you don't maintain
    \item
      You work on your branch
    \item
      When you are ready to integrate your work, rebase your work onto
      \textcolor{blue}{\texttt{origin/master}}
    \end{itemize}
  \item
    The maintainer

    \begin{itemize}
    \tightlist
    \item
      Does not have to do any integration work
    \item
      Does just a fast forward or a clean apply (no conflicts)
    \end{itemize}
  \end{itemize}
\end{itemize}

\begin{frame}{Golden Rule of Rebasing}
\protect\phantomsection\label{golden-rule-of-rebasing}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Remember}}: rebasing means abandoning existing
  commits and creating new ones that are similar but different
\end{itemize}

\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Problem}}

  \begin{itemize}
  \tightlist
  \item
    You push commits to a remote
  \item
    Others pull commits and base work on them
  \item
    You rewrite commits with \textcolor{blue}{\texttt{git rebase}}
  \item
    You push again with \textcolor{blue}{\texttt{git push --force}}
  \item
    Collaborators must re-merge work
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Solution}}

  \begin{itemize}
  \tightlist
  \item
    Strict: \emph{``Do not ever rebase commits outside your
    repository''}
  \item
    Loose: \emph{``Rebase your branch if only you use it, even if pushed
    to a server''}
  \end{itemize}
\end{itemize}

\begin{frame}[fragile]{Rebase vs Merge: Philosophical Considerations}

  \end{itemize}

  their preference

  \begin{itemize}

  \item

    \tightlist
    \item
      Even if you have pushed but you know the branch is yours
    \item

    \item

  \item
    Only \textcolor{blue}{\texttt{git merge}} to master to preserve the
    history of how something was built
  \end{itemize}

  \tightlist
  \item
    I like to squash-and-merge branches to

  \end{itemize}

\begin{frame}[fragile]{Remote Branches}
\protect\phantomsection\label{remote-branches}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Remote branches}} are pointers to branches in
  remote repositories \begingroup \small \color{blue}

\begin{verbatim}
> git remote -v
origin  git@github.com:gpsaggese/umd_classes.git (fetch)
origin  git@github.com:gpsaggese/umd_classes.git (push)
\end{verbatim}

  \endgroup
\item
  \textbf{\textcolor{blue}{Tracking branches}}

  \begin{itemize}
  \tightlist
  \item
    Local references representing the state of the remote repository
  \item
    E.g., \textcolor{blue}{\texttt{master}} tracks
    \textcolor{blue}{\texttt{origin/master}}
  \item
    You can't change the remote branch (e.g.,
    \textcolor{blue}{\texttt{origin/master}})
  \item
    You can change the tracking branch (e.g.,
    \textcolor{blue}{\texttt{master}})
  \item
    Git updates tracking branches when you do
    \textcolor{blue}{\texttt{git fetch origin}} (or
    \textcolor{blue}{\texttt{git pull}})
  \end{itemize}
\item
  To share code in a local branch you need to push it to a remote
  \begingroup \small \color{blue}

\begin{verbatim}
> git push origin serverfix
\end{verbatim}

  \endgroup
\item
  To work on it \begingroup \small \color{blue}

\begin{verbatim}
> git checkout -b serverfix origin/serverfix
\end{verbatim}

  \endgroup
\end{itemize}
\end{frame}

  \end{itemize}

  \begin{itemize}

  \item
    ``Forking'' a repo
  \item

    \begin{itemize}
    \tightlist
    \item
      Clone the repo and create a branch with their work
    \item
      Create a writable fork of the project
    \item
      Push branches to the fork
    \item
      Prepare a PR (Pull Request) with their work
    \end{itemize}
  \item
    Project maintainer:

    \begin{itemize}
    \tightlist
    \item
      Reviews PRs
    \item
      Accepts PRs
    \item
      Integrates PRs
    \end{itemize}
  \item
    In practice, it's the project maintainer who pulls the code when
    it's ready, instead of external contributors pushing the code
  \end{itemize}

    E.g., Linux, GitHub (forking) workflow
  \end{itemize}
\end{itemize}

\begin{enumerate}
\tightlist
\item
  \textbf{\textcolor{red}{One repo is the official project}}

  \begin{itemize}
  \tightlist
  \item
    Only the project maintainer pushes to the public repo
  \item
    E.g., \textcolor{blue}{\texttt{causify-ai/csfy}}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Each contributor}}

  \begin{itemize}
  \tightlist
  \item
    Has read access to everyone else's public repo
  \item
    Forks the project into a private copy

    \begin{itemize}
    \tightlist
    \item
      Write access to their own public repo
    \item
      E.g., \textcolor{blue}{\texttt{gpsaggese/csfy}}
    \end{itemize}
  \item
    Makes changes
  \item
    Pushes changes to their own public copy
  \item
    Sends a pull request to the maintainer asking to merge changes
  \end{itemize}
\item
  \textbf{\textcolor{blue}{The maintainer}}

  \begin{itemize}
  \tightlist
  \item
    Adds the contributor's repo as a remote
  \item
    Merges the changes into a local branch
  \item
    Tests changes locally
  \item
    Pushes the branch to the official repo
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Git log}

\item
  \textbf{\textcolor{red}{refs}} are references to:

  \begin{itemize}
  \tightlist
  \item
    \textcolor{blue}{\texttt{HEAD}} (commit you are working on, next
    commit)
  \item
    \textcolor{blue}{\texttt{origin/master}} (remote branch)
  \item
    \textcolor{blue}{\texttt{experiment}} (local branch)
  \item
    \textcolor{blue}{\texttt{d921970}} (commit)

    \textcolor{blue}{\texttt{HEAD}}, i.e., last commit

  \item

\begin{frame}{Dot notation}

\begin{columns}[T]

\tightlist
\item
  \textbf{\textcolor{red}{Double-dot notation}}

  \begin{itemize}
  \tightlist
  \item
    \textcolor{blue}{\texttt{1..2}} = commits that are reachable from 2
    but not from 1
  \item
    Like a ``difference''
  \item
    \textcolor{blue}{\texttt{git log master..experiment}} \(\to\)
    \textcolor{blue}{\texttt{D,C}}
  \item
    \textcolor{blue}{\texttt{git log experiment..master}} \(\to\)
    \textcolor{blue}{\texttt{F,E}}
  \end{itemize}

\begin{frame}{Advanced Git}
\protect\phantomsection\label{advanced-git-1}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{bisect}}

  \begin{itemize}
  \tightlist
  \item
    \textcolor{blue}{\texttt{git bisect}} helps identify the commit that
    introduced a bug

    \begin{itemize}
    \tightlist
    \item
      Bug appears at the top of the tree
    \item
      Unknown revision where it started
    \item
      Script returns 0 if good, non-zero if bad
    \item
      \textcolor{blue}{\texttt{git bisect}} finds the revision where the
      script changes from good to bad
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{filter-branch}}

  \begin{itemize}
  \tightlist
  \item
    Rewrite repository history in a scriptable way

    \begin{itemize}
    \tightlist
    \item
      E.g., change email, remove sensitive file
    \end{itemize}
  \item
    Check out each version, run a command, commit the result
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Hooks}}

  \begin{itemize}
  \tightlist
  \item
    Run scripts before committing, merging, etc
  \end{itemize}
\end{itemize}
\end{frame}
