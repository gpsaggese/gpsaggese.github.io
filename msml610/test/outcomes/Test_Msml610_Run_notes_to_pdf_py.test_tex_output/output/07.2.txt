\begin{frame}{Posterior-Based Decisions}
\protect\phantomsection\label{posterior-based-decisions}
\begin{itemize}
\tightlist
\item
  Sometimes describing the posterior is not enough

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{red}{Make decisions based on inference}}
  \end{itemize}
\item
  E.g., is the coin fair (\(\theta = 0.5\)) or biased?

  \begin{itemize}
  \tightlist
  \item
    \(\EE[\hat{\theta}] = 0.324\) suggests bias
  \item
    Can't rule out unbiased since:

    \begin{itemize}
    \tightlist
    \item
      \(HPI = [0.03, 0.65]\)
    \item
      \(0.5 \in HPI\)
    \end{itemize}
  \item
    For \textbf{\textcolor{blue}{sharper decisions}}:

    \begin{itemize}
    \tightlist
    \item
      Collect more data to reduce posterior spread
    \item
      Define a more informative prior
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Savage-Dickey Density Ratio}
\protect\phantomsection\label{savage-dickey-density-ratio}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Savage-Dickey ratio}} tests \emph{point
  null-hypotheses} in Bayesian inference
\item
  \textbf{\textcolor{darkgreen}{Idea}}: compare prior and posterior
  densities at a single point \(\theta_0\) \begingroup \scriptsize
  $$
  BF_{01} = \frac{p(\theta_0 | H_1)}{p(\theta_0 | \mathcal{D}, H_1)}
  $$
  \endgroup where:

  \begin{itemize}
  \tightlist
  \item
    \(p(\theta_0 | H_1)\) is \emph{prior} density \(\theta\) under
    alternative hypothesis \(H_1\), evaluated at \(\theta_0\)
  \item
    \(p(\theta_0 | \mathcal{D}, H_1)\) is \emph{posterior} density
    \(\theta\) under \(H_1\) evaluated at \(\theta_0\)
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Intuition}}: show how data changes belief
  about \(\theta_0\)

  \begin{itemize}
  \tightlist
  \item
    If posterior density at \(\theta_0\) is much smaller than prior
    density, strong evidence against null hypothesis \(H_0\)
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Cons}}:

  \begin{itemize}
  \tightlist
  \item
    It's a point statistics (not considering the entire posterior)
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Savage-Dickey Density Ratio: Example}
\protect\phantomsection\label{savage-dickey-density-ratio-example}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{msml610/lectures_source/figures/L07.2.Savage_Dickey.png}}
\end{column}

\begin{column}{0.45\linewidth}
\begin{itemize}
\tightlist
\item
  \(H_0\): ``coin is fair''

  \begin{itemize}
  \tightlist
  \item
    The prior is 0.87
  \item
    The posterior is 1.15
  \item
    \(BF_{10} = 0.76\) \(\to\) \textbf{\textcolor{red}{no evidence}} to
    reject the null hypothesis
  \end{itemize}
\item
  In the other case

  \begin{itemize}
  \tightlist
  \item
    \(BF_{10} = 15.06\) \(\to\)
    \textbf{\textcolor{blue}{strong evidence}} to reject the null
    hypothesis
  \end{itemize}
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Region of Practical Equivalence}
\protect\phantomsection\label{region-of-practical-equivalence}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Region of Practical Equivalence}} (ROPE) =
  interval where values are ``equivalent''
\item
  \textbf{\textcolor{darkgreen}{Example}}

  \begin{itemize}
  \tightlist
  \item
    \(H_0\): \emph{``coin is fair''} impractical if \(\theta = 0.5\)
  \item
    ROPE: \(\theta \in [0.45, 0.55]\) equivalent to \(0.5\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Hypothesis testing with ROPE and HPI}}

  \begin{itemize}
  \tightlist
  \item
    Compare ROPE with HPI (Highest-Posterior Interval)

    \begin{itemize}
    \tightlist
    \item
      HPI within ROPE \(\to\) no effect, reject \(H_1\)
    \item
      HPI outside ROPE \(\to\) effect present, reject \(H_0\)
    \item
      HPI overlaps ROPE \(\to\) inconclusive
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Decide ROPE before analysis based on domain knowledge}}

  \begin{itemize}
  \tightlist
  \item
    Picking it after analysis is like choosing p-value threshold after
    seeing p-value
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Loss Function: Motivation}
\protect\phantomsection\label{loss-function-motivation}
\begin{itemize}
\tightlist
\item
  For many problems:

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{red}{Make decisions based on inference}}
  \item
    \textbf{\textcolor{darkgreen}{Decision cost is asymmetric}}

    \begin{itemize}
    \tightlist
    \item
      Cost of a bad decision \textgreater{} or \textless{} benefit of a
      good decision
    \item
      Vaccines may cause overreaction, but benefits outweigh risks
    \end{itemize}
  \end{itemize}
\item
  Measure for \textbf{\textcolor{blue}{best decision}}:

  \begin{itemize}
  \tightlist
  \item
    Benefits of a correct decision
  \item
    Cost of a mistake
  \item
    Trade-off between benefits and costs using a loss function
  \item
    Use loss function for decisions
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Samples from Posterior Distribution}
\protect\phantomsection\label{samples-from-posterior-distribution}
\begin{itemize}
\tightlist
\item
  Given \textbf{\blue{posterior distribution $\Pr(\theta | y)$}},
  generate predictions \textbf{\red{$\tilde{y}$}} based on data
  \textbf{\green{$y$}} and estimated parameters
  \textbf{\purple{$\hat{\theta}$}}: \[
  \Pr(\textcolor{red}{\tilde{y}} | \textcolor{darkgreen}{\text{y}})
  = \int_{\textcolor{purple}{\hat{\theta}}} \Pr(\tilde{y} | \theta)
    \textcolor{blue}{\Pr(\theta | y)} d\theta
  = \int \text{model} \times \text{posterior}
  \]

  \begin{itemize}
  \tightlist
  \item
    This is called the \emph{``posterior predictive distribution''} as
    it predicts \emph{future data} using the \emph{posterior
    distribution}
  \end{itemize}
\item
  \textbf{\black{Conceptually}}

  \begin{itemize}
  \tightlist
  \item
    Sample a value of \(\theta\) from the posterior \(\Pr(\theta | y)\)
  \item
    Feed the value of \(\theta\) to the likelihood
    \(\Pr(\tilde{y} | \theta)\)
  \item
    Obtain \(\tilde{y}\)
  \end{itemize}
\item
  This process has \textbf{\black{two sources of uncertainty}}:

  \begin{itemize}
  \tightlist
  \item
    Parameter uncertainty

    \begin{itemize}
    \tightlist
    \item
      Captured by the posterior \(\Pr(\theta | y)\)
    \end{itemize}
  \item
    Sampling uncertainty

    \begin{itemize}
    \tightlist
    \item
      Captured by the likelihood \(\Pr(\tilde{y} | \theta)\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Posterior Predictive Check (PPC)}
\protect\phantomsection\label{posterior-predictive-check-ppc}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{PPC approach}}:

  \begin{itemize}
  \tightlist
  \item
    Generate predictions \(\tilde{y}\) with observed data \(y\) from
    posterior distribution
  \item
    Check consistency between predictions and observed data
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Intuition}}: can the model reproduce
  observed data?

  \begin{itemize}
  \tightlist
  \item
    Always check!
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Differences}} arise from:

  \begin{itemize}
  \tightlist
  \item
    Mistakes
  \item
    Model limitations

    \begin{itemize}
    \tightlist
    \item
      E.g., works for average behavior but fails for rare values
    \end{itemize}
  \item
    Data limitations
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bayesian Workflow Using PPC}
\protect\phantomsection\label{bayesian-workflow-using-ppc}
\begin{enumerate}
\item
  Given a process, \textbf{\textcolor{red}{true distribution}} is
  unknown / unknowable
\item
  \textbf{\textcolor{orange}{Sample the process}}

  \begin{itemize}
  \tightlist
  \item
    Get finite sample \(y\)
  \item
    E.g., experiment, survey, simulation
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Inference}}

  \begin{itemize}
  \tightlist
  \item
    Build probabilistic model using prior \(\Pr(\theta)\) and likelihood
    \(\Pr(y | \theta)\) to get posterior \(\Pr(\theta | y)\)
  \item
    Posterior is distribution of model parameters \(\theta\) given data
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Predictive distribution}}

  \begin{itemize}
  \tightlist
  \item
    Compute predictions from posterior (posterior predictive
    distribution)
  \item
    Posterior predictive is distribution of predicted samples averaged
    over posterior
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Validation}}

  \begin{itemize}
  \tightlist
  \item
    Validate model by comparing original vs predicted samples
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Student's t-distribution: Recap}
\protect\phantomsection\label{students-t-distribution-recap}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Student's t-distribution}} has 3 params:

  \begin{enumerate}
  \tightlist
  \item
    Mean \(\mu\)

    \begin{itemize}
    \tightlist
    \item
      Doesn't always exist
    \end{itemize}
  \item
    Scale \(\sigma\)

    \begin{itemize}
    \tightlist
    \item
      Similar to std dev, but doesn't always exist
    \end{itemize}
  \end{enumerate}
\end{itemize}

\begin{enumerate}
\setcounter{enumi}{2}
\tightlist
\item
  Degrees of freedom \(\nu \in [0, \infty]\)

  \begin{itemize}
  \tightlist
  \item
    ``Normality parameter'': controls distribution normality
  \item
    \(\nu=1\): heavy tails, no mean (Cauchy)
  \item
    \(\nu \to \infty\): Gaussian
  \end{itemize}
\end{enumerate}

\begin{itemize}
\tightlist
\item
  Student's t has \textbf{\textcolor{blue}{heavy tails}} (high kurtosis)

  \begin{itemize}
  \tightlist
  \item
    Values more likely far from mean compared to Normal
  \end{itemize}
\end{itemize}

\begin{frame}{Group Comparison}

  significant results between \emph{treatment} and \emph{control group}

  \begin{itemize}
  \tightlist
  \item
    \emph{How well do patients respond to a new drug vs a placebo?}

  between two groups

    well does it work?''} (estimate effect size)
  \end{itemize}

  \end{itemize}
\item
  \textbf{\textcolor{blue}{Placebo}} is a psychological phenomenon where
  a patient experiences improvements after receiving an inactive
  treatment

  \begin{itemize}
  \tightlist
  \item
    Using a placebo is better than ``no treatment''
  \item
    Shows difficulty in accounting for all factors in an experiment
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Group Comparison Bayesian-Style}
\protect\phantomsection\label{group-comparison-bayesian-style}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Frequentist approach}}

  \begin{itemize}
  \tightlist
  \item
    Compare p-value of difference of means in each group
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Bayesian approach}}

  \begin{itemize}
  \tightlist
  \item
    Compare posterior distribution of means between groups using:

    \begin{itemize}
    \tightlist
    \item
      Plot of posterior
    \item
      Cohen's d
    \item
      Probability of superiority
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Sample Size Effect}
\protect\phantomsection\label{sample-size-effect}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Sample size effect}} is the impact of the
  number of observations on statistical results (e.g., p-values,
  confidence intervals)
\item
  \textbf{\textcolor{darkgreen}{Small sample effect}}

  \begin{itemize}
  \tightlist
  \item
    Large mean difference might not be statistically significant (low
    p-value)
  \item
    Estimates (e.g., means, proportions, correlations) fluctuate widely
    due to high sampling variability
  \item
    Outliers have a disproportionate influence
  \item
    Inference is unstable: results may not replicate
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Large sample}}:

  \begin{itemize}
  \tightlist
  \item
    Tiny mean difference can be highly significant (small p-value) but
    meaningless
  \end{itemize}
\item
  E.g., Cohen's d, probability of superiority
\end{itemize}
\end{column}

\begin{frame}{Cohen's d}
\protect\phantomsection\label{cohens-d}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Cohen's d}} is the difference of means
  relative to pooled standard deviation \[
  \frac{\mu_2 - \mu_1}{\sqrt{(\sigma_1^2 + \sigma_2^2) / 2}}
  \]

  \begin{itemize}
  \tightlist
  \item
    Normalizes effect by variability for pooled std dev
  \item
    Variability of each group normalizes mean difference
  \item
    Similar to a Z-score, number of std dev values differ
    \pandocbounded{\includegraphics[keepaspectratio]{msml610/lectures_source/figures/L07.2.Cohen_d.png}}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Bayesian approach}}

  \begin{itemize}
  \tightlist
  \item
    Compute posterior distribution of means and std \(\to\) formula
  \item
    Compute distribution of Cohen's d \(\to\) summary statistics
  \end{itemize}
\end{itemize}
\end{frame}
