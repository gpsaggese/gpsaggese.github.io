\begin{frame}{Linear Model}
\protect\phantomsection\label{linear-model}
\begin{itemize}
\tightlist
\item
  Many problems can be formulated as
  \textbf{\textcolor{red}{linear regression}}:

  \begin{itemize}
  \tightlist
  \item
    \(X\) and \(Y\): uni-dimensional continuous RVs
  \item
    Dataset of paired observations: \(\{(x_1, y_1), ..., (x_n, y_n)\}\)
  \item
    \(X\): independent variables
  \item
    Model/predict dependent variable \(Y\)
  \item
    Assume linear relationship between \(Y\) and \(X\)
  \end{itemize}
\end{itemize}

\begin{frame}{Linear Model: Frequentist Approach}
\protect\phantomsection\label{linear-model-frequentist-approach}
\begin{itemize}
\item
  A \textbf{\textcolor{red}{linear model}} is described by: \[
  y = \alpha + \beta x
  \]
\item
  \textbf{\textcolor{blue}{Frequentist approach}}:

  \begin{itemize}
  \tightlist
  \item
    Find parameters \(\alpha, \beta\) using least square fitting

    \begin{itemize}
    \tightlist
    \item
      Minimize average quadratic error between observed \(y\) and
      predicted \(\hat{y}\) \[
      \text{MSE} = \frac{1}{N} \sum_{i=1}^N (\alpha + \beta x_i - \hat{y}_i)^2
      \]
    \end{itemize}
  \item
    Point-estimate from least squares corresponds to maximum
    a-posteriori with flat (uniform) priors on \(\alpha\) and \(\beta\)
  \item
    With certain assumptions, it's ``optimal''

    \begin{itemize}
    \tightlist
    \item
      Residuals are Gaussian
    \item
      Linearity in parameters
    \item
      Independence
    \item
      Homoscedasticity (constant variance of \(\varepsilon_i\))
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Linear Model: Bayesian Approach}
\protect\phantomsection\label{linear-model-bayesian-approach}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Bayesian assumptions}}:

  \begin{itemize}
  \tightlist
  \item
    Data \(y\) is Gaussian with mean \(\alpha + \beta x\) and standard
    deviation \[
    \sigma: y \sim N(\mu=\alpha + \beta x, \sigma)
    \]
  \item
    \(\alpha, \beta, \sigma\) are independent
  \item
    Priors: \begin{equation*}
      \begin{cases}
      \alpha \sim Normal(\mu_\alpha, \sigma_\alpha) \\
      \beta \sim Normal(\mu_\beta, \sigma_\beta) \\
      \sigma \sim HalfNormal(0, \sigma_\epsilon) \\
      \end{cases}
    \end{equation*}
  \end{itemize}
\item
  Use \textbf{\textcolor{blue}{vague priors}} to add problem knowledge:

  \begin{itemize}
  \tightlist
  \item
    Center intercept \(\alpha\) around \(0\)
  \item
    Info on sign of \(\beta\) sometimes available
  \item
    Use Half-Cauchy / exponential distribution for \(\sigma\)
  \item
    Use uniform prior if parameter has hard boundaries
  \end{itemize}

\begin{frame}{Generalized Liner Model (GLM)}
\protect\phantomsection\label{generalized-liner-model-glm}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Generalization of linear model}} using
  different distributions for likelihood: \begin{equation*}
    \begin{cases}
    \alpha \sim prior \\
    \beta \sim prior \\
    \mu = \alpha + \beta X \\
    \theta \sim prior \\
    Y \sim \phi(f(\mu), \theta) \\
    \end{cases}
  \end{equation*} where:

  \begin{itemize}
  \tightlist
  \item
    \(\phi\): arbitrary distribution

    \begin{itemize}
    \tightlist
    \item
      E.g., Normal, Student's t, negative binomial
    \end{itemize}
  \item
    \(\theta\): auxiliary parameter for \(\phi\)

    \begin{itemize}
    \tightlist
    \item
      E.g., \(\sigma\) for Normal
    \end{itemize}
  \item
    \(f()\): ``inverse link function'' transforming \(\mu\) from real
    line to domain of \(\phi\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Overdispersion}
\protect\phantomsection\label{overdispersion}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Overdispersion}} occurs when variance of count
  data is much larger than mean (e.g., 15-20x)
\item
  \textbf{\textcolor{darkgreen}{Examples}}:

  \begin{itemize}
  \tightlist
  \item
    \emph{Epidemiology}: modeling disease outbreak, number of new
    infections fluctuates widely
  \item
    \emph{Customer service}: large variation in daily complaints
  \end{itemize}
\item
  \textbf{\textcolor{blue}{How to model it?}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Use Poisson}:

    \begin{itemize}
    \tightlist
    \item
      \includegraphics[width=0.125in,height=\textheight,keepaspectratio]{assets/emoji/wrong.png}:
      mean and variance must be equal
    \end{itemize}
  \item
    \emph{Use Gaussian}:

    \begin{itemize}
    \tightlist
    \item
      Mismatch predicting negative rented bikes
    \item
      Poor fit on positive side
    \end{itemize}
  \item
    \emph{Use Negative Binomial}:

    \begin{itemize}
    \tightlist
    \item
      Better fit, though not perfect
    \item
      Right tail predictions differ, but high demand probability is low
    \item
      Second parameter controls variance
    \item
      \includegraphics[width=0.125in,height=\textheight,keepaspectratio]{assets/emoji/check_mark.png}:
      overall better than Normal model
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Classification with Logistic Regression}

  \item

  \item

  \end{itemize}
\end{itemize}

\begin{frame}{Boundary Decision for a Classifier}
\protect\phantomsection\label{boundary-decision-for-a-classifier}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Boundary decision}} \(\delta\) for a
  classifier are the values of independent variables making output
  probability 0.5
\item
  E.g., for logistic regression \(\delta\): \begin{align*}
    & 0.5 = logit(\alpha + \beta \delta) \\
    & 0 = \alpha + \beta \delta\\
    & \delta = - \frac{\alpha}{\beta} \\
    \end{align*}

  \begin{itemize}
  \tightlist
  \item
    Decision boundary uncertainty due to \(\alpha\) and \(\beta\)
    uncertainty
  \end{itemize}
\item
  Probability threshold \(0.5\) for equal
  \textbf{\textcolor{blue}{misclassification risk}}

  \begin{itemize}
  \tightlist
  \item
    Misclassification cost may not be symmetrical
  \item
    Minimize false negatives

    \begin{itemize}
    \tightlist
    \item
      E.g., \emph{``patient has disease, disease missed''}
    \end{itemize}
  \item
    Minimize false positives

    \begin{itemize}
    \tightlist
    \item
      E.g., \emph{``patient doesn't have disease, predicted''}
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Odds}
\protect\phantomsection\label{odds}
\begin{itemize}
\item
  The \textbf{\textcolor{red}{odds of event}} \emph{``\(y=1\)''} is the
  ratio of favorable to unfavorable events \[
  \text{odds} \defeq \frac{\Pr(y=1)}{1 - \Pr(y=1)}
  \]
\item
  Transformation between probability, odds, and log-odds

  \begin{itemize}
  \tightlist
  \item
    More intuitive in ``gambling''
  \item
    E.g., odds of rolling a 2 on a fair die are
    \(\frac{1/6}{5/6} = \frac{1}{5} = 0.2\)

    \begin{itemize}
    \tightlist
    \item
      One favorable event for 5 unfavorable events
    \end{itemize}
  \end{itemize}
\item
  Interpreting
  \textbf{\textcolor{blue}{logistic regression in terms of odds}}:

  \begin{itemize}
  \tightlist
  \item
    Since \(\theta = logit(\alpha + \beta x)\) and
    \(\theta = \Pr(y = 1)\): \[
    \alpha + \beta x = \log ( \frac{\Pr(y=1)}{1 - \Pr(y=1)} )
    \]
  \item
    Logistic regression models log-odds linearly
  \item
    \(\beta\) is the increase in log-odds for a unit change in \(x\)
  \end{itemize}
\end{itemize}
\end{frame}
