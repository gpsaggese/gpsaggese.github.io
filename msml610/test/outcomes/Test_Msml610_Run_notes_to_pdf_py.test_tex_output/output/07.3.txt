\begin{frame}{Hierarchical Models}
\protect\phantomsection\label{hierarchical-models}
\begin{itemize}
\item
  Aka ``multilevel'', ``nested'', ``mixed-effects'' models
\item
  \textbf{\textcolor{red}{Key observation}}: data points share
  structure, but also have variations

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{orange}{Group data}}

    \begin{itemize}
    \tightlist
    \item
      E.g., sales in cities: each city is a market, with common trends
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Hierarchical structure}}

    \begin{itemize}
    \tightlist
    \item
      E.g., students in a school: each student is different, with common
      factors
    \end{itemize}
  \item
    \textbf{\textcolor{teal}{Repeated measurements on same objects}}
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Idea}}:
  \includegraphics[width=0.14583in,height=\textheight,keepaspectratio]{assets/emoji/lightbulb.png}

  \begin{itemize}
  \tightlist
  \item
    Model shares information between groups, but allows differences
  \item
    Parameters of prior distributions have a prior distribution

    \begin{itemize}
    \tightlist
    \item
      Aka ``hyper-priors'' (!)
    \end{itemize}
  \end{itemize}
\item
  You can't do this with frequentist approach, only Bayesian approach
\end{itemize}
\end{frame}

\begin{frame}{Hierarchical Models: Examples}
\protect\phantomsection\label{hierarchical-models-examples}
\begin{itemize}
\item
  Many data problems lend themselves to
  \textbf{\textcolor{red}{hierarchical descriptions}}
\item
  E.g.,

  \begin{itemize}
  \tightlist
  \item
    Medical research:

    \begin{itemize}
    \tightlist
    \item
      Estimate drug effectiveness
    \item
      Categorize patients by demographics, disease severity
    \item
      Estimate cure probability for subgroups
    \end{itemize}
  \item
    Market research

    \begin{itemize}
    \tightlist
    \item
      Understand consumer purchasing behavior
    \item
      Categorize consumers by age, gender, income, education
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Unpooled, Pooled, Hierarchical Models}
\protect\phantomsection\label{unpooled-pooled-hierarchical-models}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Pooled}}

  \begin{itemize}
  \tightlist
  \item
    Groups have the same priors
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Unpooled}}

  \begin{itemize}
  \tightlist
  \item
    Groups have different priors
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Hierarchical}}

  \begin{itemize}
  \tightlist
  \item
    Groups have different priors which come from a common prior
  \end{itemize}
\end{itemize}

  Given experimental measures of chemical shifts vs theoretical values,
  evaluate model using different styles

  \begin{enumerate}
  \tightlist
  \item
    \textbf{\textcolor{red}{Pooled}}

    \begin{itemize}
    \tightlist
    \item
      Compute difference between estimates and measures, fit Gaussian
    \item
      More accurate estimates / lose amino acid info
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Unpooled}}

    \begin{itemize}
    \tightlist
    \item
      Fit 20 Gaussians for 20 amino acids
    \item
      Detailed analysis / less accuracy
    \end{itemize}
  \item
    \textbf{\textcolor{blue}{Hierarchical}}

    \begin{itemize}
    \tightlist
    \item
      Model groups assuming common population
    \end{itemize}
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Shrinkage}
\protect\phantomsection\label{shrinkage}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Hierarchical models shrink parameters towards a common mean}}

  \begin{itemize}
  \tightlist
  \item
    Groups share information through the hyper-prior
  \item
    Model groups as neither independent nor a single group
  \item
    Less responsive to extreme values in individual groups
  \item
    Improve estimation for small groups using data from others
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.4\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{msml610/lectures_source/figures/L07.3.Shrinkage.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Amount of shrinkage}} depends on data:

  \begin{itemize}
  \tightlist
  \item
    Groups with more data influence estimates more
  \item
    Similar groups reinforce common estimation
  \item
    Global optimization
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Result}}: inference is more stable
\end{itemize}
\end{frame}

\begin{frame}{You Need to Know When to Stop}
\protect\phantomsection\label{you-need-to-know-when-to-stop}
\begin{itemize}
\tightlist
\item
  You can create
  \textbf{\textcolor{red}{hierarchical models with as many levels}} as
  you want

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Pros}}:

    \begin{itemize}
    \tightlist
    \item
      Leverage data structure
    \end{itemize}
  \item
    \textbf{\textcolor{blue}{Cons}}

    \begin{itemize}
    \tightlist
    \item
      Don't improve inference quality
    \item
      Complicate interpretation
    \end{itemize}
  \end{itemize}
\item
  \emph{``Add as many degrees as freedom as needed, but not more than
  what is warranted''} (Occam's razor)
\end{itemize}
\end{frame}
